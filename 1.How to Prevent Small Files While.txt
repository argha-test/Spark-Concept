from pyspark.sql import SparkSession
from pyspark.sql.functions import col, sum as _sum, when, current_timestamp

spark = SparkSession.builder.appName("Sales_Transformation").getOrCreate()

# 1. Read data from ADLS (Delta/Parquet/CSV)
df = spark.read.format("csv") \
    .option("header", "true") \ 
	.option("inferschema","true")\
    .load("/mnt/adls/raw/sales_data.csv")

# 2. Data Cleaning & Transformation
df_clean = df.withColumn("sales_amount", col("sales_amount").cast("double")) \
    .withColumn("region", when(col("region").isNull(), "Unknown").otherwise(col("region"))) \
    .withColumn("ingestion_time", current_timestamp())

# 3. Aggregation
df_agg = df_clean.groupBy("region") \
    .agg(_sum("sales_amount").alias("total_sales"))

# 4. Write to Delta Table
df_agg.write.format("delta") \
    .mode("overwrite") \ 
	.partition by ("Date")
    .save("/mnt/adls/processed/sales_delta")

# Optional: Register as table
spark.sql("""
    CREATE TABLE IF NOT EXISTS sales_summary
    USING DELTA
    LOCATION '/mnt/adls/processed/sales_delta'
""")
